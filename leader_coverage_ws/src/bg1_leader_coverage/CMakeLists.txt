cmake_minimum_required(VERSION 3.16)
project(bg1_leader_coverage)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(bg1_leader_coverage_interfaces REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
set(LEADER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/leader/coverage_path_planner" CACHE PATH "Leader source directory (read-only)")
set(PORT "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(GLOB_RECURSE LEADER_INPUTS CONFIGURE_DEPENDS "${LEADER_SOURCE}/*.h" "${LEADER_SOURCE}/*.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${LEADER_INPUTS} tools/prepare_core.py)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_core.py ${LEADER_SOURCE} ${PORT}
                COMMAND_ERROR_IS_FATAL ANY)
add_library(leader_core STATIC
 ${PORT}/jsoncpp.cpp ${PORT}/curve.cpp ${PORT}/task_helper.cpp ${PORT}/node.cpp
 ${PORT}/CoveragePathPlanner.cpp ${PORT}/BezierTrajectoryGeneratorWaypoint.cpp
 ${PORT}/FreeSpaceCalculator.cpp ${PORT}/A_star_pathplanner.cpp ${PORT}/A_star_pathplanner_fast.cpp
 src/adapter.cpp)
target_include_directories(leader_core PUBLIC src ${PORT} ${OpenCV_INCLUDE_DIRS})
target_compile_options(leader_core PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/src/compat.hpp)
ament_target_dependencies(leader_core rclcpp nav_msgs geometry_msgs nav2_costmap_2d bg1_leader_coverage_interfaces)
target_link_libraries(leader_core ${OpenCV_LIBS} Eigen3::Eigen yaml-cpp)
add_executable(planner_server src/server.cpp)
target_link_libraries(planner_server leader_core)
install(TARGETS planner_server DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/region_selector.py scripts/region_command.py scripts/coverage_motion.py scripts/motion_command.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/selection.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(FILES ${PORT}/source_hashes.json DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 find_package(ament_cmake_pytest REQUIRED)
 ament_add_gtest(test_planner test/test_planner.cpp SKIP_LINKING_MAIN_LIBRARIES)
 target_link_libraries(test_planner leader_core)
 ament_add_pytest_test(motion_protocol test/test_motion_protocol.py ENV ROS_DOMAIN_ID=96 TIMEOUT 45)
 ament_add_pytest_test(motion_command test/test_motion_command.py)
 ament_add_pytest_test(motion_node test/test_motion_node.py ENV ROS_DOMAIN_ID=95)
 ament_add_pytest_test(selection test/test_selection.py)
 ament_add_pytest_test(selector_node test/test_selector_node.py ENV ROS_DOMAIN_ID=93)
endif()
ament_package()
